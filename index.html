9. الجزء العملي (Practical Implementation)

تم بناء نموذج كشف التهديدات باستخدام مكتبة TensorFlow لتحليل تدفق البيانات الحيوية للنظام:

# Deep Learning Detection Script
import tensorflow as tf
from tensorflow.keras.layers import LSTM, Dense

def build_sentinel_model():
    model = tf.keras.Sequential([
        LSTM(64, input_shape=(100, 1), return_sequences=True),
        LSTM(32),
        Dense(1, activation='sigmoid') # Malicious or Safe
    ])
    model.compile(optimizer='adam', loss='binary_crossentropy')
    return model
