const fs = require('fs');
const path = require('path');

const vault = './secure_data';
fs.readdirSync(vault).forEach(file => {
    const filePath = path.join(vault, file);

    const encrypted = Buffer.from(content).toString('base64');

    fs.renameSync(filePath, filePath + '.safe');
    
    console.log(`[MONITOR] Attack Pattern Blocked: ${file}`);
});
