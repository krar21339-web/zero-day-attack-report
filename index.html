01 // PHILOSOPHY

الرؤية الاستراتيجية لأنظمة
الاستدراج الرقمي

إن مشروع ESP32 Honeypot يمثل تحولاً جوهرياً في كيفية التعامل مع التهديدات السيبرانية المحلية. الهدف من هذا العمل هو ليس فقط بناء جدار حماية، بل إنشاء "محور ذكاء" يقوم بجمع البيانات من المهاجمين في الوقت الفعلي.

لقد تم تصميم النظام ليحاكي نقاط الوصول الضعيفة (Vulnerable Access Points)، مما يجبر المهاجم على استخدام أدواته البرمجية لكشف الثغرات، وفي تلك اللحظة، يقوم نظامنا بالتقاط "البصمة الرقمية" (Digital Fingerprinting) للمهاجم وتخزينها للتحليل الجنائي اللاحق.

02 // HARDWARE ARCHITECTURE

بنية المكونات المادية
وتوزيع الدوائر الإلكترونية

03 // SOURCE CODE ANALYSIS

المنطق البرمجي المتطور
لإدارة الهجمات

تمت برمجة النظام باستخدام لغة C++ مع توظيف تقنيات الـ Asynchronous التفاعلية لضمان عدم حدوث عنق زجاجة (Bottleneck) أثناء معالجة البيانات الجنائية.

// Advanced Forensic Logging System v1.0
#include <WiFi.h>
#include <DNSServer.h>
#include <Wire.h>

const byte DNS_PORT = 53;
IPAddress apIP(172, 0, 0, 1);
DNSServer dnsServer;

void initializeCyberShield() {
  Serial.begin(115200);
  WiFi.mode(WIFI_AP);
  WiFi.softAPConfig(apIP, apIP, IPAddress(255, 255, 255, 0));
  WiFi.softAP("FREE_UNIVERSITY_WIFI"); 
  
  dnsServer.start(DNS_PORT, "*", apIP);
  Serial.println("[SYSTEM_READY]: Monitoring all ingress packets...");
}

void loop() {
  dnsServer.processNextRequest();
  if (checkNewIntruder()) {
    captureForensicMetadata();
    updateOLEDInterface();
  }
}
